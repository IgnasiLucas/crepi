ASM=../2024-10-04/radloci.fa
BAM=../2024-10-04/BAM

if [ ! -d z ]; then mkdir z; fi
for i in 2 3 4 5 6 7 8 9 10; do
   if [ ! -e z/minCov$i.sfs ]; then
      if [ ! -e z/minCov$i.saf.idx ]; then
         angsd -i $BAM/L019Cxa.bam \
               -anc $ASM \
               -doGlf 1 \
               -GL 1 \
               -doSaf 1 \
               -C 50 \
               -ref $ASM \
               -minQ 20 \
               -minmapQ 30 \
               -setMinDepthInd $i \
               -out z/minCov$i 2> /dev/null
      fi
      realSFS -fold 1 \
              z/minCov$i.saf.idx \
              1> z/minCov$i.sfs \
              2> /dev/null
   fi
done
find z -name '*.gz'  -delete
find z -name '*.idx' -delete
sfs <- lapply(2:10, function(x) scan(paste0('z/minCov', x, '.sfs'), quiet = TRUE))
pi  <- sapply(sfs, function(x) x[2] / sum(x))
N   <- sapply(sfs, sum)
cov <- 2:10
plot(cov, pi)

plot(N, pi)

plot(cov, N)
